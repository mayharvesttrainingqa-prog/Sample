<script>
(() => {
  const STORAGE_KEY = "sample_online_records_v1";
  let records = [];
  let editingId = null;
  const $ = id => document.getElementById(id);
  const form = $("recordForm");

  try { records = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]"); }
  catch { records = []; }

  function save() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(records));
    render();
  }
  function safe(value) {
    return String(value ?? "").replace(/[&<>"']/g, ch => ({
      "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;"
    }[ch]));
  }
  function render() {
    const query = $("search").value.trim().toLowerCase();
    const filtered = records.filter(r =>
      [r.name, r.department, r.remarks].some(v => String(v).toLowerCase().includes(query))
    );
    $("totalCount").textContent = records.length;
    $("lastUpdated").textContent = records.length
      ? new Date(Math.max(...records.map(r => r.updatedAt || r.createdAt))).toLocaleString()
      : "—";
    $("recordsBody").innerHTML = filtered.map((r, i) => `
      <tr>
        <td>${i + 1}</td><td>${safe(r.name)}</td><td>${safe(r.department || "—")}</td>
        <td>${safe(r.remarks || "—")}</td><td>${new Date(r.createdAt).toLocaleDateString()}</td>
        <td><div class="actions">
          <button class="secondary" data-edit="${safe(r.id)}" type="button">Edit</button>
          <button class="danger" data-delete="${safe(r.id)}" type="button">Delete</button>
        </div></td>
      </tr>`).join("");
    $("emptyState").hidden = filtered.length > 0;
  }

  form.addEventListener("submit", event => {
    event.preventDefault();
    const name = $("person").value.trim();
    if (!name) return;
    const now = Date.now();
    if (editingId) {
      const item = records.find(r => r.id === editingId);
      if (item) Object.assign(item, {
        name, department: $("department").value.trim(),
        remarks: $("remarks").value.trim(), updatedAt: now
      });
    } else {
      records.unshift({
        id: (crypto.randomUUID ? crypto.randomUUID() : String(now) + Math.random()),
        name, department: $("department").value.trim(),
        remarks: $("remarks").value.trim(), createdAt: now, updatedAt: now
      });
    }
    resetForm();
    save();
  });

  $("recordsBody").addEventListener("click", event => {
    const editId = event.target.dataset.edit;
    const deleteId = event.target.dataset.delete;
    if (editId) {
      const item = records.find(r => r.id === editId);
      if (!item) return;
      editingId = editId;
      $("person").value = item.name;
      $("department").value = item.department || "";
      $("remarks").value = item.remarks || "";
      $("formTitle").textContent = "Edit Record";
      $("saveButton").textContent = "Update Record";
      $("cancelButton").hidden = false;
      $("person").focus();
      window.scrollTo({top: 0, behavior: "smooth"});
    }
    if (deleteId) {
      const item = records.find(r => r.id === deleteId);
      if (item && confirm(`Delete the record for ${item.name}?`)) {
        records = records.filter(r => r.id !== deleteId);
        if (editingId === deleteId) resetForm();
        save();
      }
    }
  });

  function resetForm() {
    form.reset();
    editingId = null;
    $("formTitle").textContent = "Add New Record";
    $("saveButton").textContent = "Save Record";
    $("cancelButton").hidden = true;
  }
  $("cancelButton").addEventListener("click", resetForm);
  $("search").addEventListener("input", render);
  render();
})();
</script>
